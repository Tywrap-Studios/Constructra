    Configuring Constructra

    If you happen to be stuck, or don't like a particular part of the gameplay, you are probably happy to hear this mod is highly configurable!

    There are two types of config files:

    • Server Side
    • Client Side

    Both of these files can be found in the run/config directory, and are json5 files, meaning they support comments, which aim to be as informative as possible

    Info

    Changes on either end will only affect that end, meaning that a changed made on the server won't affect the client side configs, and vice versa.

    Live examples:

    If you want a live example of the configs being used, take a gander at the dev environment configurations!

    /* (1)
       This file has all the configuration options for Constructra on the client side.
       Changes here will not affect a dedicated server (nor other players online)!

       The following value is irrelevant, do not change it, though.
    */
    "run_info": "None",
    /* The Unit System to use for Measurements.
       Choices: METRIC, FREEDOM
       Type: String (from set)
       Default: METRIC
    */
    "measure_units": "METRIC",
    // Config related to your Play Time Safety.
    "play_time_safety": {
        /* Whether to send 2020 reminders.
           After 20 minutes of playing, look away for 20 seconds.
           Type: Boolean
           Default: false
        */
        "send_2020_reminders": false,
        /* The amount of minutes between 2020 reminders.
           Only applies if send_2020_reminders is true.
           Type: Integer
           Range: >0-60
           Default: 20
        */
        "interval": 20,
        /* Whether to send reminders to take small breaks after long whiles of playing.
           Type: Boolean
           Default: false
        */
        "send_break_reminders": true
    }
}

    {
    /* (2)
       This file has all the configuration options for Constructra.
       Do not change the config_version unless you know what you're doing or else some of your changes may be arbitrarily reset and we will use the default values.
    */
    "config_version": "1.0",
    // Config related to Resources.
    "resources": {
        /* The default purity of each Resource in your world.
           Choices: NONE, IMPURE, NORMAL, PURE or RANDOM.
           Type: String (from Set)
           Default: RANDOM
        */
        "default_resource_purity": "RANDOM",
        /* Whether Nodes can be obstructed. When set to false, Nodes don't need to be destructed first in order to be harvested from.
           Type: Boolean
           Default: true
        */
        "do_obstructions": true,
        /* The amount of times you have to consecutively harvest from a Node to be able to de-obstruct it.
           Only applies if do_obstructions is true.
           Type: Integer
           Default: 3
        */
        "consecutive_harvests_for_destruction": 3,
        // Certain Modifiers for ResourceNode Purity Related Mechanics.
        "purity_modifiers": {
            /* Whether the Purity of a ResourceNode affects at which rate the resource is given.
               If false is chosen, we will use the rate of a NORMAL node, so feel free to tweak that value.
               ResourceNodes will still have their Purity displayed!
               Type: Boolean
               Default: true
            */
            "does_purity_affect_rate": true,
            /* The Rate at which a PURE node should give its Resource.
               Type: Double (in seconds)
               Default: 0.8
            */
            "pure": 0.8,
            /* The Rate at which a NORMAL node should give its Resource.
               Type: Double (in seconds)
               Default: 1.2
            */
            "normal": 1.2,
            /* The Rate at which an IMPURE node should give its Resource.
               Type: Double (in seconds)
               Default: 1.9
            */
            "impure": 1.9
        },
        /* Whether to display certain visual hints as to what block is the exact ResourceNode Centre. e.g. through the use of particles.
           Type: Boolean
           Default: false
        */
        "visualize_centres": true
    },
    // Config related to Commands.
    "commands": {
        /* The command alias for the Constructra command. Change this if you're having compatibility issues with other mods.
           Type: String
           Default: ca
        */
        "command_alias": "ca",
        /* The permission level required to use the Node commands in general.
           Type: Integer
           Default: 2
        */
        "perm_lvl_nodes": 2,
        /* The permission level required to remove Nodes from the world. (Both nodes flush and nodes purge.)
           Note that this is a dangerous permission to give as it can break the entire world!
           Type: Integer
           Default: 4
        */
        "perm_lvl_nodes_removal": 4,
        /* The permission level required to reload the Constructra config.
           Type: Integer
           Default: 2
        */
        "perm_lvl_reload": 2
    },
    // Several configurations for utility features.
    "util_config": {
        // Whether to display debug information in the console.
        "debug_mode": true,
        // Whether to suppress all warnings from this mod. NOT RECOMMENDED.
        "suppress_warns": false
    }
}
